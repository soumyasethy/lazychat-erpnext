```python
# Cycle 13.2 — defensive sanitizer for agent-generated Page content.
# LLMs sometimes entity-escape their entire HTML output (e.g. send
# `&lt;header&gt;` when they mean `<header>`). The disk file ends up
# rendering visible literal text instead of real DOM. This helper
# detects fully-entity-encoded content (no real tags + at least one
# entity-encoded tag start) and unescapes it. Mixed content
# (real tags + intentional `&lt;` for code samples) is left alone —
# author intent wins. Apply at create_page + update_doc(Page) commit.
_ENTITY_TAG_RE = re.compile(r'&lt;[a-z!/]', re.IGNORECASE)
_REAL_TAG_RE = re.compile(r'<[a-z!/]', re.IGNORECASE)


def _decode_if_fully_entity_escaped(s):
	"""Return s with HTML entities decoded IFF s looks fully entity-escaped.

	Heuristic: contains an entity-encoded tag start (`&lt;header`) AND
	contains zero real tag starts (`<header`). Otherwise return s unchanged.
	None / empty inputs return "".
	"""
	if not s:
		return s or ""
	if _ENTITY_TAG_RE.search(s) and not _REAL_TAG_RE.search(s):
		return html.unescape(s)
	return s
```

Notes:
- Tabs (not spaces) — match the existing file indentation. Verify with `cat -A`.
- Helper is module-private (single leading underscore), no public API change.

```tsx
import { useEffect, useState } from 'react';
import { ChevronDown } from 'lucide-react';
import { useUI } from '@/store/ui';
import { useSessions } from '@/store/sessions';

const AUTO_CLEAR_MS = 30_000;

/**
 * Cycle 11 — M1 / Cycle 13.2 (session-scoped): floating "Result ready →" pill
 * that appears when an mcpPreviewAction transitions to status='done' AND its
 * card is scrolled out of view AND the action belongs to the currently active
 * session. Click → smooth-scrolls the card into view and clears `lastResult`.
 *
 * Auto-clears after 30s of staying set (covers the case where the user leaves
 * the chat tab open without acting on it). Also clears when the card re-enters
 * viewport, when the active session is different (early-bail), or when the
 * target DOM node has been unmounted (stale-clear instead of leaking a
 * pointing-at-nothing pill).
 *
 * Mounts at the MessageList level (single instance per chat).
 */
export function ResultReadyPill() {
  const lastResult = useUI((s) => s.lastResult);
  const setLastResult = useUI((s) => s.setLastResult);
  const activeId = useSessions((s) => s.activeId);
  const [offScreen, setOffScreen] = useState(false);

  // Early bail: wrong session means there's nothing for THIS session to point
  // at, even if some other session set lastResult. Don't observe DOM, don't
  // arm the auto-clear timer — just render null.
  const sessionMatches = !!lastResult && lastResult.sessionId === activeId;

  useEffect(() => {
    if (!sessionMatches) return;
    const t = setTimeout(() => setLastResult(null), AUTO_CLEAR_MS);
    return () => clearTimeout(t);
  }, [sessionMatches, setLastResult]);

  useEffect(() => {
    setOffScreen(false);
    if (!sessionMatches || !lastResult) return;
    const node = document.querySelector(
      `[data-msg-id="${CSS.escape(lastResult.msgId)}"]`,
    );
    if (!node) {
      // Stale pointer — the message DOM has been unmounted (session swap mid-
      // render, message removed, etc). Clear the state instead of leaking a
      // visible pill that can't navigate anywhere.
      setLastResult(null);
      return;
    }
    const observer = new IntersectionObserver(
      (entries) => {
        const entry = entries[0];
        if (!entry) return;
        if (entry.isIntersecting) {
          setOffScreen(false);
          setLastResult(null);
        } else {
          setOffScreen(true);
        }
      },
      { threshold: 0.1 },
    );
    observer.observe(node);
    return () => observer.disconnect();
  }, [sessionMatches, lastResult, setLastResult]);

  if (!sessionMatches || !offScreen) return null;

  function handleClick() {
    if (!lastResult) return;
    const node = document.querySelector(
      `[data-msg-id="${CSS.escape(lastResult.msgId)}"]`,
    );
    if (node) {
      (node as HTMLElement).scrollIntoView({ behavior: 'smooth', block: 'center' });
    }
    setLastResult(null);
  }

  return (
    <button
      type="button"
      data-pill="result-ready"
      onClick={handleClick}
      className="fixed bottom-20 right-4 z-40 inline-flex items-center gap-1.5 rounded-full bg-accent px-3 py-1.5 text-[12px] font-medium text-fg-on-accent shadow-lg hover:bg-accent/90 focus:outline-none focus-visible:ring-2 focus-visible:ring-ring"
    >
      <ChevronDown className="h-3.5 w-3.5" aria-hidden />
      Result ready
    </button>
  );
}
```
